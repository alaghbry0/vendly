    88→// Cheapest monthly plan (falls back to cheapest yearly → "/yr").
    89→function fromPriceInfo(product: ProductCardDTO): { cents: number; suffix: string } | null {
    90→  const active = product.plans.filter((p) => p.active);
    91→  const monthly = active.filter((p) => p.interval === "month");
    92→  if (monthly.length > 0) return { cents: Math.min(...monthly.map((p) => p.priceCents)), suffix: "/mo" };
    93→  const yearly = active.filter((p) => p.interval === "year");
    94→  if (yearly.length > 0) return { cents: Math.min(...yearly.map((p) => p.priceCents)), suffix: "/yr" };
    95→  return null;
    96→}

   256→// ---------------------------------------------------------------------------
   257→// Decorative QR (deterministic from seed) & confetti
   258→// ---------------------------------------------------------------------------
   259→
   260→function FakeQR({ seed, className }: { seed: string; className?: string }) {
   261→  const size = 21;
   262→  const cells = useMemo(() => {
   263→    let h = 2166136261;
   264→    for (let i = 0; i < seed.length; i++) {
   265→      h ^= seed.charCodeAt(i);
   266→      h = Math.imul(h, 16777619);
   267→    }
   268→    let x = h >>> 0 || 42;
   269→    const grid: boolean[] = [];
   270→    for (let i = 0; i < size * size; i++) {
   271→      x = (Math.imul(x, 1664525) + 1013904223) >>> 0;
   272→      grid.push(((x >>> 16) & 7) < 3);
   273→    }
   274→    return grid;
   275→  }, [seed]);
   276→
   277→  const finder = (r: number, c: number) => {
   278→    for (const [r0, c0] of [
   279→      [0, 0],
   280→      [0, 14],
   281→      [14, 0],
   282→    ] as const) {
   283→      const dr = r - r0;
   284→      const dc = c - c0;
   285→      if (dr >= 0 && dr <= 6 && dc >= 0 && dc <= 6) {
   286→        return dr === 0 || dr === 6 || dc === 0 || dc === 6 || (dr >= 2 && dr <= 4 && dc >= 2 && dc <= 4);
   287→      }
   288→    }
   289→    return false;
   290→  };
   291→
   292→  return (
   293→    <div className={cn("grid", className)} style={{ gridTemplateColumns: `repeat(${size}, 1fr)` }} role="img" aria-label="Payment QR code (simulated)">
   294→      {Array.from({ length: size * size }, (_, i) => {
   295→        const r = Math.floor(i / size);
   296→        const c = i % size;
   297→        const on = finder(r, c) || cells[i];
   298→        return <span key={i} className={cn("aspect-square", on ? "bg-neutral-900" : "bg-white")} />;
   299→      })}
   300→    </div>
   301→  );
   302→}

   306→function Confetti() {
   307→  const pieces = useMemo(
   308→    () =>
   309→      Array.from({ length: 32 }, (_, i) => ({
   310→        left: Math.random() * 100,
   311→        delay: Math.random() * 0.6,
   312→        duration: 1.9 + Math.random() * 1.5,
   313→        color: CONFETTI_COLORS[i % CONFETTI_COLORS.length],
   314→        size: 6 + Math.random() * 7,
   315→        rotate: Math.random() * 360,
   316→        round: Math.random() > 0.5,
   317→      })),
   318→    []
   319→  );
   320→  return (
   321→    <div aria-hidden className="pointer-events-none absolute inset-0 overflow-hidden">
   322→      {pieces.map((p, i) => (
   323→        <motion.span
   324→          key={i}
   325→          className="absolute -top-3"
   326→          style={{
   327→            left: `${p.left}%`,
   328→            width: p.size,
   329→            height: p.round ? p.size : p.size * 0.45,
   330→            backgroundColor: p.color,
   331→            borderRadius: p.round ? 999 : 2,
   332→          }}
   333→          initial={{ y: -16, opacity: 1, rotate: 0 }}
   334→          animate={{ y: 340, opacity: [1, 1, 0], rotate: p.rotate + 200 }}
   335→          transition={{ duration: p.duration, delay: p.delay, ease: "easeIn" }}
   336→        />
   337→      ))}
   338→    </div>
   339→  );

   454→function DiscoverView() {
   455→  const params = useAppStore((s) => s.params);
   456→  const nonce = useAppStore((s) => s.nonce);
   457→  const { toast } = useToast();
   458→
   459→  const [searchText, setSearchText] = useState(params.query ?? "");
   460→  const [category, setCategory] = useState(params.category ?? "ALL");
   461→  const [sort, setSort] = useState("featured");
   462→  const [products, setProducts] = useState<ProductCardDTO[] | null>(null);
   463→  const [loading, setLoading] = useState(true);
   464→
   465→  const gridRef = useRef<HTMLDivElement>(null);
   466→  const howRef = useRef<HTMLDivElement>(null);
   467→
   468→  // Sync with external navigation params (e.g. header search).
   469→  useEffect(() => {
   470→    setSearchText(params.query ?? "");
   471→    setCategory(params.category ?? "ALL");
   472→  }, [params.query, params.category]);
   473→
   474→  // Debounced catalog fetch, driven by filters + global refresh nonce.
   475→  useEffect(() => {
   476→    let cancelled = false;
   477→    const t = setTimeout(async () => {
   478→      setLoading(true);
   479→      try {
   480→        const qs = new URLSearchParams();
   481→        const q = searchText.trim();
   482→        if (q) qs.set("q", q);
   483→        if (category !== "ALL") qs.set("category", category);
   484→        if (sort !== "featured") qs.set("sort", sort);
   485→        const suffix = qs.toString();
   486→        const res = await api<{ products: ProductCardDTO[] }>(`/api/products${suffix ? `?${suffix}` : ""}`);
   487→        if (!cancelled) setProducts(res.products);
   488→      } catch (e) {
   489→        if (!cancelled) {
   490→          setProducts([]);
   491→          toast({ title: "Could not load products", description: (e as Error).message, variant: "destructive" });
   492→        }
   493→      } finally {
   494→        if (!cancelled) setLoading(false);
   495→      }
   496→    }, 320);
   497→    return () => {
   498→      cancelled = true;
   499→      clearTimeout(t);
   500→    };
   501→  }, [searchText, category, sort, nonce, toast]);
   502→

   770→function ReviewDialog({
   771→  product,
   772→  open,
   773→  onOpenChange,
   774→  onSubmitted,
   775→}: {
   776→  product: ProductDetailDTO;
   777→  open: boolean;
   778→  onOpenChange: (v: boolean) => void;
   779→  onSubmitted: () => void;
   780→}) {
   781→  const { toast } = useToast();
   782→  const [rating, setRating] = useState(5);
   783→  const [comment, setComment] = useState("");
   784→  const [busy, setBusy] = useState(false);
   785→
   786→  useEffect(() => {
   787→    if (open) {
   788→      setRating(5);
   789→      setComment("");
   790→    }
   791→  }, [open]);
   792→
   793→  async function submit() {
   794→    if (comment.trim().length < 4) {
   795→      toast({ title: "Review too short", description: "Write at least a few words about your experience.", variant: "destructive" });
   796→      return;
   797→    }
   798→    setBusy(true);
   799→    try {
   800→      await api(`/api/products/${product.id}/reviews`, { json: { rating, comment: comment.trim() } });
   801→      toast({ title: "Review posted", description: "Thanks for sharing your experience!" });
   802→      onOpenChange(false);
   803→      onSubmitted();
   804→    } catch (e) {
   805→      toast({ title: "Could not post review", description: (e as Error).message, variant: "destructive" });
   806→    } finally {
   807→      setBusy(false);
   808→    }
   809→  }
   810→
   811→  return (
   812→    <Dialog open={open} onOpenChange={onOpenChange}>
   813→      <DialogContent className="sm:max-w-md">
   814→        <DialogHeader>
   815→          <DialogTitle>Review {product.title}</DialogTitle>
   816→          <DialogDescription>Share your experience to help other buyers decide.</DialogDescription>
   817→        </DialogHeader>
   818→        <div className="space-y-5">
   819→          <div>
   820→            <Label className="text-sm font-medium">Your rating</Label>
   821→            <RadioGroup value={String(rating)} onValueChange={(v) => setRating(Number(v))} className="mt-2.5 flex gap-1">
   822→              {[1, 2, 3, 4, 5].map((i) => (
   823→                <div key={i} className="relative flex items-center">
   824→                  <RadioGroupItem value={String(i)} id={`star-${i}`} className="peer sr-only" />
   825→                  <Label
   826→                    htmlFor={`star-${i}`}
   827→                    aria-label={`${i} star${i > 1 ? "s" : ""}`}
   828→                    className="cursor-pointer rounded-lg p-2 transition-colors hover:bg-muted peer-focus-visible:ring-2 peer-focus-visible:ring-ring"
   829→                  >
   830→                    <Star className={cn("h-7 w-7 transition-colors", i <= rating ? "fill-amber-400 text-amber-400" : "text-muted-foreground/30")} />
   831→                  </Label>
   832→                </div>
   833→              ))}
   834→            </RadioGroup>
   835→          </div>
   836→          <div>
   837→            <Label htmlFor="review-comment" className="text-sm font-medium">
   838→              Your review
   839→            </Label>
   840→            <Textarea
   841→              id="review-comment"
   842→              value={comment}
   843→              onChange={(e) => setComment(e.target.value)}
   844→              rows={4}
   845→              maxLength={600}
   846→              placeholder="What did you like? What could be better?"
   847→              className="mt-2"
   848→            />
   849→            <p className="mt-1 text-right text-xs text-muted-foreground">{comment.length}/600</p>
   850→          </div>
   851→        </div>
   852→        <DialogFooter>
   853→          <Button variant="outline" onClick={() => onOpenChange(false)} disabled={busy}>

   858→            Post review
   859→          </Button>
   860→        </DialogFooter>
