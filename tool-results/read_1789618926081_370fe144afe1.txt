   164→function monthlyCents(plan: PlanDTO): number {

   168→function shortWallet(addr: string): string {
   169→  return addr.length > 12 ? `${addr.slice(0, 6)}…${addr.slice(-4)}` : addr;
   170→}

   172→function pmShort(pm: PaymentMethodDTO): string {
   173→  if (pm.type === "CARD") return `${pm.brand || "Card"} ••••${pm.last4}`;
   174→  if (pm.type === "PAYPAL") return pm.email || "PayPal account";
   175→  return shortWallet(pm.walletAddress || "");
   176→}

   178→function maskKey(key: string): string {
   179→  const parts = key.split("-");
   180→  return parts.map((p, i) => (i === 0 || i === parts.length - 1 ? p : "•".repeat(Math.max(4, p.length)))).join("-");
   181→}
   182→
   183→function fileIcon(mime: string): LucideIcon {
   184→  if (mime.startsWith("image/")) return FileImage;
   185→  if (mime.startsWith("video/")) return FileVideo;
   186→  if (mime.startsWith("audio/")) return FileAudio;
   187→  if (mime.includes("zip") || mime.includes("compressed") || mime.includes("tar")) return FileArchive;
   188→  if (mime.includes("json") || mime.includes("javascript") || mime.includes("html")) return FileCode;
   189→  return FileText;
   190→}

   192→/** One-line summary of the next billing event for a subscription. */
   193→function nextChargeLine(sub: SubscriptionDTO): string {
   194→  if (sub.status === "PAST_DUE") return "Payment failed — update payment method";
   195→  if (sub.status === "CANCELED") return `Ended ${fmtDate(sub.currentPeriodEnd)}`;
   196→  if (sub.status === "TRIALING") return `Trial ends ${fmtDate(sub.trialEndsAt)}`;
   197→  if (sub.cancelAtPeriodEnd) return `Cancels ${fmtDate(sub.currentPeriodEnd)}`;
   198→  return `Next charge ${fmtMoney(sub.plan.priceCents)} on ${fmtDate(sub.currentPeriodEnd)}`;
   199→}

   201→function sortSubs(subs: SubscriptionDTO[]): SubscriptionDTO[] {
   202→  return [...subs].sort(
   203→    (a, b) =>
   204→      (STATUS_ORDER[a.status] ?? 9) - (STATUS_ORDER[b.status] ?? 9) ||
   205→      +new Date(a.currentPeriodEnd) - +new Date(b.currentPeriodEnd)
   206→  );
   207→}

   222→function usePortalData() {
   223→  const nonce = useAppStore((s) => s.nonce);
   224→  const [data, setData] = useState<PortalData | null>(null);
   225→  const [error, setError] = useState<string | null>(null);
   226→
   227→  useEffect(() => {
   228→    if (!useAppStore.getState().user) return;
   229→    let alive = true;
   230→    const load = async () => {
   231→      try {
   232→        const [subsRes, invoicesRes, methodsRes, licensesRes, grantsRes] = await Promise.all([
   233→          api<{ subscriptions: SubscriptionDTO[] }>("/api/subscriptions"),
   234→          api<{ invoices: InvoiceDTO[] }>("/api/invoices"),
   235→          api<{ paymentMethods: PaymentMethodDTO[] }>("/api/payment-methods"),
   236→          api<{ licenses: LicenseDTO[] }>("/api/licenses"),
   237→          api<{ grants: GrantDTO[] }>("/api/grants"),
   238→        ]);
   239→        if (!alive) return;
   240→        setData({
   241→          subs: subsRes.subscriptions,
   242→          invoices: invoicesRes.invoices,
   243→          methods: methodsRes.paymentMethods,
   244→          licenses: licensesRes.licenses,
   245→          grants: grantsRes.grants,
   246→        });
   247→        setError(null);
   248→      } catch (e) {
   249→        if (alive) setError(e instanceof Error ? e.message : "Failed to load your hub.");
   250→      }
   251→    };
   252→    void load();
   253→    return () => {
   254→      alive = false;
   255→    };
   256→  }, [nonce]);
   257→
   258→  return { data, error, loading: !data && !error };
   259→}

   913→function SubscriptionsTab({
   914→  data,
   915→  reload,
   916→  goToInvoices,
   917→}: {
   918→  data: PortalData;
   919→  reload: () => void;
   920→  goToInvoices: (subId: string) => void;
   921→}) {
   922→  const { toast } = useToast();
   923→  const navigate = useAppStore((s) => s.navigate);
   924→  const { subs, methods } = data;
   925→
   926→  const [busyKey, setBusyKey] = useState<string | null>(null);
   927→  const [tierSub, setTierSub] = useState<SubscriptionDTO | null>(null);
   928→  const [tierPlans, setTierPlans] = useState<PlanDTO[] | null>(null);
   929→  const [tierPlanId, setTierPlanId] = useState<string | null>(null);
   930→  const [cancelMode, setCancelMode] = useState<{ sub: SubscriptionDTO; mode: "period_end" | "now" } | null>(null);
   931→  const plansCache = useRef<Record<string, PlanDTO[]>>({});
   932→
   933→  // Invalidate plan cache whenever subscription data refreshes.
   934→  useEffect(() => {
   935→    plansCache.current = {};
   936→  }, [subs]);
   937→
   938→  function openTier(sub: SubscriptionDTO) {
   939→    setTierPlanId(null);
   940→    setTierPlans(plansCache.current[sub.product.id] ?? null);
   941→    setTierSub(sub);
   942→  }
   943→
   944→  function closeTier() {
   945→    setTierSub(null);
   946→    setTierPlans(null);
   947→    setTierPlanId(null);
   948→  }
   949→
   950→  // Fetch plans for the "Change tier" dialog when a cache miss was staged.
   951→  useEffect(() => {
   952→    if (!tierSub || tierPlans) return;
   953→    const pid = tierSub.product.id;
   954→    let alive = true;
   955→    const load = async () => {
   956→      try {
   957→        const res = await api<{ product: ProductDetailDTO }>(`/api/products/${pid}`);
   958→        if (!alive) return;
   959→        plansCache.current[pid] = res.product.plans;
   960→        setTierPlans(res.product.plans);
   961→      } catch {
   962→        if (!alive) return;
   963→        toast({ title: "Couldn't load plans", description: "Please try again.", variant: "destructive" });
   964→        closeTier();
   965→      }
   966→    };
   967→    void load();
   968→    return () => {
   969→      alive = false;
   970→    };
   971→  }, [tierSub, tierPlans, toast]);
   972→
   973→  async function runSubAction(
   974→    sub: SubscriptionDTO,
   975→    action: string,
   976→    extra: Record<string, unknown> = {},
   977→    successTitle = "Subscription updated"
   978→  ): Promise<boolean> {
   979→    setBusyKey(`${sub.id}:${action}`);
   980→    try {
   981→      const res = await api<{ ok: boolean; message: string }>(`/api/subscriptions/${sub.id}`, {
   982→        method: "PATCH",
   983→        json: { action, ...extra },
   984→      });
   985→      toast({ title: successTitle, description: res.message });
   986→      reload();
   987→      return true;
   988→    } catch (e) {
   989→      toast({
   990→        title: "Action failed",
   991→        description: e instanceof Error ? e.message : "Please try again.",
   992→        variant: "destructive",
   993→      });
   994→      return false;
   995→    } finally {
   996→      setBusyKey(null);
   997→    }
   998→  }
   999→
  1000→  async function changePaymentMethod(sub: SubscriptionDTO, pmId: string) {
  1001→    if (pmId === sub.paymentMethod?.id) return;
  1002→    setBusyKey(`${sub.id}:update_payment`);
  1003→    try {
  1004→      const res = await api<{ ok: boolean; message: string }>(`/api/subscriptions/${sub.id}`, {
  1005→        method: "PATCH",
  1006→        json: { action: "update_payment", paymentMethodId: pmId },
  1007→      });
  1008→      toast({
  1009→        title: "Payment method updated",
  1010→        description:
  1011→          sub.status === "PAST_DUE"
  1012→            ? `${res.message} The next billing run will retry the failed charge automatically.`
  1013→            : res.message,
  1014→      });
  1015→      reload();
  1016→    } catch (e) {
  1017→      toast({
  1018→        title: "Couldn't update payment method",
  1019→        description: e instanceof Error ? e.message : "Please try again.",
  1020→        variant: "destructive",
  1021→      });
  1022→    } finally {
  1023→      setBusyKey(null);
  1024→    }
  1025→  }
  1026→
  1027→  const activeCount = subs.filter((s) => s.status === "ACTIVE").length;
  1028→  const trialCount = subs.filter((s) => s.status === "TRIALING").length;
  1029→  const sorted = useMemo(() => sortSubs(subs), [subs]);
  1030→
  1031→  return (
  1032→    <motion.section variants={stagger} initial="hidden" animate="show" className="space-y-5">
  1033→      <motion.div variants={fadeUp}>
  1034→        <SectionHeader
  1035→          title="Memberships"
  1036→          description={`${activeCount} active · ${trialCount} trialing · ${subs.length - activeCount - trialCount} ended`}
  1037→        />
  1038→      </motion.div>
  1039→
  1040→      {sorted.map((sub) => (
  1041→        <motion.div key={sub.id} variants={fadeUp}>
  1042→          <SubscriptionCard
  1043→            sub={sub}
  1044→            methods={methods}
  1045→            busy={!!busyKey && busyKey.startsWith(`${sub.id}:`)}
  1046→            onAction={runSubAction}
  1047→            onChangePayment={changePaymentMethod}
  1048→            onOpenTier={() => openTier(sub)}
  1049→            onCancel={(mode) => setCancelMode({ sub, mode })}
  1050→            onInvoices={() => goToInvoices(sub.id)}
  1051→          />
  1052→        </motion.div>
  1053→      ))}
  1054→
  1055→      {subs.length === 0 && (
  1056→        <motion.div variants={fadeUp}>
  1057→          <EmptyState
  1058→            icon={Repeat}
  1059→            title="No memberships yet"
  1060→            description="Browse the marketplace and join your first community or grab a digital product."
